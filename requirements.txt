# Grid engine dependencies (Person 1)

# Only pandapower's own version (and pytest, for running tests) is pinned.
# pandas/numpy/scipy are deliberately left UNPINNED: pandapower declares
# different transitive version constraints on these depending on your
# Python version (e.g. it requires scipy~=1.16 on Python 3.11 but
# scipy~=1.18 on Python 3.12). Hard-pinning exact versions here (as an
# earlier version of this file did) broke `pip install` for anyone on a
# different Python version than the one the pins were taken from -- let
# pip resolve pandas/numpy/scipy from pandapower's own requirements
# instead, whatever they are for your interpreter.
pandapower==3.5.4
pytest==9.1.1
